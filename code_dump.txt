# Helper function to classify domain groups
def get_domain_group(col: str) -> str:
    c = col.lower()
    if any(k in c for k in ["walk", "stairs", "chair", "grip", "mobility"]):
        return "mobility"
    elif any(k in c for k in ["bp", "pressure", "systolic", "diabetes", "stroke", "heart", "hiv", "cd4", "disease"]):
        return "disease"
    elif any(k in c for k in ["bmi", "weight", "height"]):
        return "bmi"
    elif any(k in c for k in ["memory", "recall", "cog", "depress", "sad", "social", "friend", "cognition"]):
        return "cognition"
    else:
        return "other"

# Helper function to compute computed features
def fill_computed_features(df: pd.DataFrame, feature_columns: list[str]) -> pd.DataFrame:
    df = df.copy()
    
    domain_map = {}
    for col in feature_columns:
        if "__" in col:
            domain = col.split("__")[0]
            domain_map.setdefault(domain, []).append(col)
            
    for d, cols in domain_map.items():
        block = df[cols].apply(pd.to_numeric, errors="coerce")
        df[f"{d}_mean"] = block.mean(axis=1)
        if f"{d}_std" in feature_columns:
            df[f"{d}_std"] = block.std(axis=1) if len(cols) > 1 else 0.0
        df[f"{d}_missing_frac"] = block.isna().mean(axis=1)
        
        row_median = block.median(axis=1).iloc[0]
        if pd.isna(row_median):
            df[f"{d}_impairment"] = 0
        else:
            df[f"{d}_impairment"] = (block.mean(axis=1) < row_median).astype(int)
        df[f"{d}_count"] = block.notna().sum(axis=1)
        
    df["global_mean"] = df[raw_cols].mean(axis=1)
    df["global_missing"] = df[raw_cols].isna().mean(axis=1)
    global_std = df[raw_cols].std(axis=1)
    df["health_index"] = df["global_mean"] / (1 + global_std)
    
    from features.frailty_index import compute_frailty_index
    df["frailty_index"] = compute_frailty_index(df)
    
    df = df[feature_columns]
    return df

from __future__ import annotations

from typing import Iterable
import numpy as np
import pandas as pd

from config.haalsi_config import MISSING_VALUES


def clean_missing_values(df: pd.DataFrame) -> pd.DataFrame:
    """
    Safe replacement of HAALSI missing codes (-97, -98, -99)
    avoiding pandas block corruption
    """

    out = df.copy()

    for col in out.columns:
        try:
            out[col] = out[col].replace(MISSING_VALUES, np.nan)
        except Exception:
            continue

    return out


def standardize_column_names(df: pd.DataFrame) -> pd.DataFrame:
    out = df.copy()
    out.columns = [str(c).strip() for c in out.columns]
    return out


def safe_numeric(series: pd.Series) -> pd.Series:
    return pd.to_numeric(series, errors="coerce")


def binary_from_any(series: pd.Series) -> pd.Series:
    s = safe_numeric(series)
    return (s > 0).astype("float64")


def clip_percentile(series: pd.Series, lower: float = 0.01, upper: float = 0.99) -> pd.Series:
    s = safe_numeric(series)
    lo, hi = s.quantile([lower, upper])
    return s.clip(lo, hi)

from __future__ import annotations

from collections import defaultdict
from typing import Dict, List
import pandas as pd


def get_wave_columns(df: pd.DataFrame) -> Dict[str, List[str]]:
    wave_cols = defaultdict(list)
    for col in df.columns:
        s = str(col)
        if s.startswith("W1"):
            wave_cols["W1"].append(s)
        elif s.startswith("W2"):
            wave_cols["W2"].append(s)
        elif s.startswith("W3"):
            wave_cols["W3"].append(s)
    return dict(wave_cols)


def split_by_wave(df: pd.DataFrame) -> dict[str, pd.DataFrame]:
    cols = get_wave_columns(df)
    return {wave: df[wave_cols].copy() for wave, wave_cols in cols.items()}


def get_panel_pairs(df: pd.DataFrame, base_wave: str = "W1", follow_wave: str = "W2") -> tuple[pd.DataFrame, pd.DataFrame]:
    base_cols = [c for c in df.columns if str(c).startswith(base_wave)]
    follow_cols = [c for c in df.columns if str(c).startswith(follow_wave)]
    return df[base_cols].copy(), df[follow_cols].copy()

from __future__ import annotations

from typing import Any
import numpy as np
import pandas as pd
import shap
from sklearn.calibration import CalibratedClassifierCV
from sklearn.pipeline import Pipeline


def explain_decline(model: Any, X: pd.DataFrame) -> tuple[list[str], dict[str, float]]:
    """
    Computes SHAP values for the decline model.
    Handles CalibratedClassifierCV (averaging over folds) and direct Pipeline estimators.
    
    Returns:
        - top_10_features: list of the top 10 features by mean absolute SHAP value.
        - feature_contributions: dictionary of feature names to their SHAP values for the first sample.
    """
    if isinstance(model, CalibratedClassifierCV):
        all_shap_values = []
        for calibrated_classifier in model.calibrated_classifiers_:
            estimator = calibrated_classifier.estimator  # Pipeline
            
            # Preprocess features
            preprocessor = Pipeline(estimator.steps[:-1])
            X_trans = preprocessor.transform(X)
            X_trans_df = pd.DataFrame(X_trans, columns=X.columns)
            
            clf = estimator.steps[-1][1]  # XGBClassifier
            explainer = shap.TreeExplainer(clf)
            exp = explainer(X_trans_df)
            shap_vals = exp.values
            
            # Handle binary class dimension
            if shap_vals.ndim == 3:
                shap_vals = shap_vals[:, :, 1]
            all_shap_values.append(shap_vals)
            
        mean_shap_values = np.mean(all_shap_values, axis=0)
    else:
        # Direct Pipeline or base estimator
        estimator = model
        preprocessor = Pipeline(estimator.steps[:-1])
        X_trans = preprocessor.transform(X)
        X_trans_df = pd.DataFrame(X_trans, columns=X.columns)
        
        clf = estimator.steps[-1][1]
        explainer = shap.TreeExplainer(clf)
        exp = explainer(X_trans_df)
        mean_shap_values = exp.values
        if mean_shap_values.ndim == 3:
            mean_shap_values = mean_shap_values[:, :, 1]

    # Mean absolute SHAP values across the batch
    mean_abs_shap = np.mean(np.abs(mean_shap_values), axis=0)
    feature_importance = pd.Series(mean_abs_shap, index=X.columns)
    top_10_features = feature_importance.sort_values(ascending=False).head(10).index.tolist()
    
    # Feature contributions for the first row
    first_sample_shap = mean_shap_values[0]
    feature_contributions = dict(zip(X.columns, first_sample_shap))
    
    return top_10_features, feature_contributions

from __future__ import annotations

import re
from pathlib import Path
from typing import Any

import pandas as pd

from config.haalsi_config import CROSSWALK_PATH


def _normalize_col(col: Any) -> str:
    return re.sub(r"\s+", " ", str(col).strip().lower())


def load_crosswalk(path: str | Path = CROSSWALK_PATH) -> pd.DataFrame:
    path = Path(path)
    if not path.exists():
        raise FileNotFoundError(f"Crosswalk file not found: {path}")

    xls = pd.ExcelFile(path)
    frames = []

    for sheet in xls.sheet_names:
        df = pd.read_excel(path, sheet_name=sheet)
        if df.empty:
            continue
        df = df.copy()
        df.columns = [_normalize_col(c) for c in df.columns]
        df["source_sheet"] = sheet
        frames.append(df)

    if not frames:
        raise ValueError(f"No readable sheets found in crosswalk: {path}")

    crosswalk = pd.concat(frames, ignore_index=True)
    crosswalk = crosswalk.dropna(how="all")
    return crosswalk


def guess_columns(crosswalk: pd.DataFrame) -> dict[str, str | None]:
    cols = list(crosswalk.columns)

    def find_contains(tokens: list[str]) -> str | None:
        for c in cols:
            cc = c.lower()
            if all(t in cc for t in tokens):
                return c
        return None

    return {
        "w1": find_contains(["wave", "1"]) or find_contains(["w1"]) or find_contains(["wave 1"]),
        "w2": find_contains(["wave", "2"]) or find_contains(["w2"]) or find_contains(["wave 2"]),
        "w3": find_contains(["wave", "3"]) or find_contains(["w3"]) or find_contains(["wave 3"]),
        "question": find_contains(["question", "text"]) or find_contains(["question"]),
        "label": find_contains(["label"]),
        "concept": find_contains(["concept"]) or find_contains(["topic"]),
        "varname": find_contains(["varname"]) or find_contains(["variable", "name"]),
    }


def clean_crosswalk(crosswalk: pd.DataFrame) -> pd.DataFrame:
    out = crosswalk.copy()
    out.columns = [_normalize_col(c) for c in out.columns]
    out = out.dropna(how="all")
    return out


def build_variable_registry(crosswalk: pd.DataFrame) -> pd.DataFrame:
    crosswalk = clean_crosswalk(crosswalk)
    cols = guess_columns(crosswalk)

    if cols["varname"] is None and cols["question"] is None:
        raise ValueError(
            "Could not detect the expected crosswalk columns. "
            "Inspect the workbook and update `guess_columns()`."
        )

    out = crosswalk.copy()

    if cols["question"] is None:
        out["question_text"] = ""
    else:
        out["question_text"] = out[cols["question"]].fillna("").astype(str)

    if cols["label"] is None:
        out["label_text"] = ""

    if cols["concept"] is None:

def build_domain_variable_lists(df: pd.DataFrame) -> dict[str, dict[str, list[str]]]:
    """
    Robust mapping using actual dataset columns (NOT crosswalk)
    """

    domain_varlists = {}

    for col in df.columns:

        # -----------------------
        # Detect wave
        # -----------------------
        if col.startswith("W1"):
            wave = "W1"
        elif col.startswith("W2"):
            wave = "W2"
        elif col.startswith("W3"):
            wave = "W3"
        else:
            continue

        # -----------------------
        # Detect domain
        # -----------------------
        domain = detect_domain(col)

        if domain is None:
            continue

        # -----------------------
        # Initialize
        # -----------------------
        if domain not in domain_varlists:
            domain_varlists[domain] = {
                "W1": [],
                "W2": [],
                "W3": []
            }

        domain_varlists[domain][wave].append(col)

    return domain_varlists


# ---------------------------

def build_feature_table(
    df: pd.DataFrame,
    domain_varlists: dict[str, dict[str, list[str]]],
    wave: str = "W1"
) -> pd.DataFrame:

    frames = []

    for domain, waves in domain_varlists.items():

        vars_ = waves.get(wave, [])

        if not vars_:
            continue

        cols = [v for v in vars_ if v in df.columns]

        if not cols:
            continue

        subset = df[cols].copy()

        # Prefix with domain for clarity
        subset = subset.add_prefix(f"{domain}__")

        frames.append(subset)

    if not frames:
        return pd.DataFrame(index=df.index)

    return pd.concat(frames, axis=1)


# ---------------------------
# PAIRWISE WAVE FRAMES (DECLINE)
# ---------------------------

def pairwise_domain_frames(
    df: pd.DataFrame,
    domain_varlists: dict[str, dict[str, list[str]]],
    wave_base: str = "W1",
    wave_follow: str = "W2"
):

    base_frames = []
    follow_frames = []

    for domain, waves in domain_varlists.items():

        base_vars = waves.get(wave_base, [])
        follow_vars = waves.get(wave_follow, [])

        for base_var in base_vars:
            follow_var = base_var.replace(wave_base, wave_follow, 1)

            if base_var in df.columns and follow_var in df.columns:
                base_frames.append(
                    df[[base_var]].rename(
                        columns={base_var: f"{domain}__{base_var}"}
                    )
                )

                follow_frames.append(
                    df[[follow_var]].rename(
                        columns={follow_var: f"{domain}__{follow_var}"}
                    )
                )

    base = pd.concat(base_frames, axis=1) if base_frames else pd.DataFrame(index=df.index)
    follow = pd.concat(follow_frames, axis=1) if follow_frames else pd.DataFrame(index=df.index)

    return base, follow

def compute_frailty_index(feature_frame: pd.DataFrame) -> pd.Series:
    """
    Compute a simple deficit-count frailty index over selected HAALSI domains.
    The input should already contain numeric columns mapped from the crosswalk.
    """

    if feature_frame.empty:
        return pd.Series(dtype="float64")

    deficits = []

    for col in feature_frame.columns:
        series = pd.to_numeric(feature_frame[col], errors="coerce")

        # Generic deficit rule:
        # - binary / ordinal adverse indicators: >0 counts as a deficit
        # - if the column is mostly continuous, we use a robust z-style fallback below
        valid = series.dropna()
        if valid.empty:
            continue

        nunique = valid.nunique(dropna=True)

        if nunique <= 5:
            deficit = (series > 0).astype(float)
        else:
            median = valid.median()
            mad = (valid - median).abs().median()
            scale = mad if mad and mad > 0 else valid.std(ddof=0)
            if not scale or np.isnan(scale) or scale == 0:
                deficit = pd.Series(0.0, index=series.index)
            else:
                z = (series - median) / scale
                deficit = (z > 1.0).astype(float)

        deficits.append(deficit.rename(col))

    if not deficits:
        return pd.Series(0.0, index=feature_frame.index, name="frailty_index")

    deficit_df = pd.concat(deficits, axis=1)
    frailty = deficit_df.mean(axis=1).fillna(0.0)
    frailty.name = "frailty_index"
    return frailty

def compute_combined_risk(decline_model, trajectory_model, label_encoder, X):

    # -----------------------
    # Decline probability
    # -----------------------
    decline_probs = decline_model.predict_proba(X)[:, 1]

    # -----------------------
    # Trajectory prediction
    # -----------------------
    traj_score = trajectory_model.predict(X)

    # normalize between 0 and 1
    traj_score = (traj_score - traj_score.min()) / (traj_score.max() - traj_score.min())

    # -----------------------
    # Combine risk
    # -----------------------
    combined = 0.6 * decline_probs + 0.4 * traj_score

    # Normalize
    combined = np.clip(combined, 0, 1)

    return pd.DataFrame({
        "decline_risk": decline_probs,
        "trajectory_score": traj_score,
        "combined_risk": combined
    })

def build_model(random_state: int = RANDOM_STATE) -> Pipeline:
    base = XGBClassifier(
        n_estimators=250,
        max_depth=4,
        learning_rate=0.05,
        subsample=0.85,
        colsample_bytree=0.85,
        reg_lambda=1.0,
        objective="binary:logistic",
        eval_metric="logloss",
        random_state=random_state,
        tree_method="hist",
    )

    # XGBoost handles missing values directly, but the imputer keeps the pipeline explicit.
    model = Pipeline(
        steps=[
            ("imputer", SimpleImputer(strategy="median")),
            ("scaler", StandardScaler(with_mean=False)),
            ("clf", base),
        ]
    )
    return model


def calibrate_model(model: Pipeline, X_train: pd.DataFrame, y_train: pd.Series) -> CalibratedClassifierCV:
    calibrator = CalibratedClassifierCV(model, method="sigmoid", cv=3)
    calibrator.fit(X_train, y_train)
    return calibrator


def train_decline_model(
    X: pd.DataFrame,
    y: pd.Series,
    test_size: float = 0.2,
    random_state: int = RANDOM_STATE,
) -> TrainedModelBundle:
    feature_columns = list(X.columns)

    X_train, X_test, y_train, y_test = train_test_split(
        X,
        y,
        test_size=test_size,
        stratify=y if y.nunique() > 1 else None,
        random_state=random_state,
    )

    model = build_model(random_state=random_state)
    calibrated = calibrate_model(model, X_train, y_train)

    proba = calibrated.predict_proba(X_test)[:, 1]
    auc = roc_auc_score(y_test, proba) if y_test.nunique() > 1 else float("nan")
    brier = brier_score_loss(y_test, proba) if len(np.unique(y_test)) > 1 else float("nan")

    preds = (proba >= 0.5).astype(int)
    report = classification_report(y_test, preds, output_dict=True, zero_division=0)

    metrics = {
        "auc": float(auc) if auc == auc else np.nan,
        "brier": float(brier) if brier == brier else np.nan,
        "precision": float(report.get("1", {}).get("precision", np.nan)),
        "recall": float(report.get("1", {}).get("recall", np.nan)),
        "f1": float(report.get("1", {}).get("f1-score", np.nan)),
    }

    return TrainedModelBundle(model=calibrated, feature_columns=feature_columns, metrics=metrics)

from __future__ import annotations

import numpy as np
import pandas as pd

from config.haalsi_config import AGE_WEIGHT, DECLINE_WEIGHT, FRAILTY_WEIGHT


def _normalize_age(age: pd.Series | float | int) -> pd.Series:
    if isinstance(age, (float, int)):
        return pd.Series([age], dtype="float64")
    return pd.to_numeric(age, errors="coerce").fillna(age.median() if hasattr(age, "median") else 0)


def fuse_risk(frailty: pd.Series, decline_prob: pd.Series, age: pd.Series) -> pd.Series:
    age_s = pd.to_numeric(age, errors="coerce")
    age_scaled = (age_s - age_s.min()) / (age_s.max() - age_s.min() + 1e-9)
    frailty_s = pd.to_numeric(frailty, errors="coerce").fillna(0.0).clip(0, 1)
    decline_s = pd.to_numeric(decline_prob, errors="coerce").fillna(0.0).clip(0, 1)

    risk = (
        FRAILTY_WEIGHT * frailty_s +
        DECLINE_WEIGHT * decline_s +
        AGE_WEIGHT * age_scaled
    ).clip(0, 1)
    risk.name = "fused_risk"
    return risk


def categorize_risk(risk: pd.Series) -> pd.Series:
    bins = [-np.inf, 0.30, 0.55, 0.75, np.inf]
    labels = ["Low", "Moderate", "High", "Very High"]
    return pd.cut(risk, bins=bins, labels=labels, right=False)

import numpy as np
import pandas as pd
from sklearn.mixture import GaussianMixture
from sklearn.preprocessing import StandardScaler


def cluster_trajectories(features: pd.DataFrame, n_clusters: int = 3):
    """
    Fit a standardized Gaussian mixture model to temporal trajectory features.
    Returns cluster labels and a serializable model bundle.
    """
    if features.empty:
        raise ValueError("Trajectory feature matrix is empty.")

    clean = features.replace([np.inf, -np.inf], np.nan)
    clean = clean.fillna(clean.median(numeric_only=True))

    scaler = StandardScaler()
    scaled = scaler.fit_transform(clean)

    gmm = GaussianMixture(
        n_components=n_clusters,
        covariance_type="diag",
        reg_covar=1e-6,
        random_state=42,
    )

    labels = gmm.fit_predict(scaled)

    model_bundle = {
        "scaler": scaler,
        "gmm": gmm,
        "feature_columns": list(clean.columns),
        "n_clusters": n_clusters,
    }

    return labels, model_bundle

def build_training_matrices(df, domain_varlists):

    # -----------------------
    # BASELINE FEATURES (W1 ONLY)
    # -----------------------
    X = build_feature_table(df, domain_varlists, wave="W1")

    if X.empty:
        raise ValueError("No W1 features found.")

    # -----------------------
    # NUMERIC CLEAN
    # -----------------------
    for c in X.columns:
        X[c] = pd.to_numeric(X[c], errors="coerce")

    # -----------------------
    # DOMAIN GROUPING
    # -----------------------
    domain_map = {}
    for col in X.columns:
        if "__" not in col:
            continue
        domain = col.split("__")[0]
        domain_map.setdefault(domain, []).append(col)

    # -----------------------
    # DOMAIN-LEVEL FEATURES (HIGH IMPACT)
    # -----------------------
    for d, cols in domain_map.items():

        block = X[cols]

        # Mean health
        X[f"{d}_mean"] = block.mean(axis=1)

        # Variability within domain
        if len(cols) > 1:
            X[f"{d}_std"] = block.std(axis=1)

        # Missingness
        X[f"{d}_missing_frac"] = block.isna().mean(axis=1)

        # Impairment proxy (below median)
        median_val = block.median().median()
        X[f"{d}_impairment"] = (block.mean(axis=1) < median_val).astype(int)

        X[f"{d}_count"] = block.notna().sum(axis=1)

    # -----------------------
    # GLOBAL FEATURES
    # -----------------------
    X["global_mean"] = X.mean(axis=1)
    X["global_missing"] = X.isna().mean(axis=1)

    global_std = X.std(axis=1)
    X["health_index"] = X["global_mean"] / (1 + global_std)

    # Frailty
    X["frailty_index"] = compute_frailty_index(X)

    # -----------------------
    # INTERACTIONS (VERY IMPORTANT)
    # -----------------------
    if "age_proxy" in X.columns:
        X["age_frailty"] = X["age_proxy"] * X["frailty_index"]

    # -----------------------
    # FOLLOW-UP (ONLY FOR LABEL)
    # -----------------------
    follow = build_feature_table(df, domain_varlists, wave="W2")

    y_decline = compute_decline_label(X, follow)

    # -----------------------
    # TRAJECTORY
    # -----------------------
    y_trajectory, traj_cluster_model = build_trajectory_labels(df, domain_varlists)

    # -----------------------
    # FINAL CLEAN
    # -----------------------
    X = X.replace([np.inf, -np.inf], np.nan)
    X = X.fillna(X.median(numeric_only=True))

    y_decline = y_decline.reindex(X.index).fillna(0).astype(int)
    y_trajectory = y_trajectory.reindex(X.index)

    return X, y_decline, y_trajectory, traj_cluster_model

from __future__ import annotations

import pandas as pd
import numpy as np

from targets.trajectory_utils import compute_domain_score

def print_domain_availability(domain_varlists):
    print("\n[DEBUG] Domain availability:")
    for d in domain_varlists:
        w1 = len(domain_varlists[d].get("W1", []))
        w2 = len(domain_varlists[d].get("W2", []))
        w3 = len(domain_varlists[d].get("W3", []))
        print(f"{d:20s} | W1: {w1:3d} | W2: {w2:3d} | W3: {w3:3d}")


def zscore_series(s: pd.Series) -> pd.Series:
    s = pd.to_numeric(s, errors="coerce")
    mu = s.mean()
    sigma = s.std(ddof=0)
    if pd.isna(sigma) or sigma == 0:
        return pd.Series(0.0, index=s.index)
    return (s - mu) / sigma


def compute_multidomain_scores(df: pd.DataFrame, domain_varlists):
    """
    Build a composite longitudinal signal from domains that are present in W1, W2, and W3.
    Each domain score is z-scored before averaging so one scale cannot dominate the others.
    """
    preferred_domains = ["mobility", "disease", "cognition", "bmi"]

    used_domains = []
    w1_scores = []
    w2_scores = []
    w3_scores = []

    print("\n[INFO] Domains used for trajectory:")

    for d in preferred_domains:
        w1_cols = domain_varlists.get(d, {}).get("W1", [])
        w2_cols = domain_varlists.get(d, {}).get("W2", [])
        w3_cols = domain_varlists.get(d, {}).get("W3", [])

        if w1_cols and w2_cols and w3_cols:
            used_domains.append(d)
            print(f" - {d}")

            w1_scores.append(zscore_series(compute_domain_score(df, w1_cols)))
            w2_scores.append(zscore_series(compute_domain_score(df, w2_cols)))
            w3_scores.append(zscore_series(compute_domain_score(df, w3_cols)))

    if not used_domains:
        raise ValueError("No valid domains found for trajectory modeling.")

    weights = {
        "mobility": 2.0,
        "disease": 1.5,
        "bmi": 1.0
    }

    w1_list, w2_list, w3_list = [], [], []

    for d in used_domains:
        weight = weights.get(d, 1.0)

        w1_list.append(weight * compute_domain_score(df, domain_varlists[d]["W1"]))
        w2_list.append(weight * compute_domain_score(df, domain_varlists[d]["W2"]))
        w3_list.append(weight * compute_domain_score(df, domain_varlists[d]["W3"]))

    w1 = pd.concat(w1_list, axis=1).mean(axis=1)
    w2 = pd.concat(w2_list, axis=1).mean(axis=1)
    w3 = pd.concat(w3_list, axis=1).mean(axis=1)

    return w1, w2, w3, used_domains


def extract_trajectory_features(w1: pd.Series, w2: pd.Series, w3: pd.Series) -> pd.DataFrame:
    delta1 = w2 - w1
    delta2 = w3 - w2
    slope = (w3 - w1) / 2.0
    acceleration = delta2 - delta1
    variability = pd.concat([w1, w2, w3], axis=1).std(axis=1)

    return pd.DataFrame(
        {
            "delta1": delta1,
            "delta2": delta2,
            "slope": slope,
            "acceleration": acceleration,
            "variability": variability,
        }
    )


def label_clusters(traj_features: pd.DataFrame, labels) -> tuple[pd.Series, dict[int, str]]:
    """
    Map latent clusters to clinical labels using rank order of temporal trend.
    Higher trajectory score = worse progression.
    """
    df = traj_features.copy()
    df["cluster"] = labels

    summary = df.groupby("cluster")[["delta1", "delta2", "slope", "acceleration", "variability"]].mean()

    summary["trajectory_score"] = (
        0.60 * summary["slope"]
        + 0.25 * summary["acceleration"]
        + 0.15 * summary["variability"]
    )

    ordered_clusters = summary["trajectory_score"].sort_values(ascending=False).index.tolist()

    label_names = ["Rapid Decline", "Slow Decline", "Stable"]
    label_map: dict[int, str] = {}

    for i, cluster_id in enumerate(ordered_clusters):
        label_map[int(cluster_id)] = label_names[min(i, len(label_names) - 1)]

    print("\n[INFO] Cluster interpretation:")
    for k in sorted(label_map):
        print(f"Cluster {k} → {label_map[k]}")

    labeled_series = pd.Series(labels, index=traj_features.index).map(label_map)
    return labeled_series, label_map


def build_trajectory_labels(df: pd.DataFrame, domain_varlists):
    print_domain_availability(domain_varlists)

    w1, w2, w3, used_domains = compute_multidomain_scores(df, domain_varlists)

    traj_features = extract_trajectory_features(w1, w2, w3)
    traj_features = traj_features.replace([np.inf, -np.inf], np.nan)
    traj_features = traj_features.fillna(traj_features.median(numeric_only=True))

    labels, model = cluster_trajectories(traj_features)

    # -----------------------
    # STABLE TRAJECTORY SCORE
    # -----------------------

    # Primary signal: long-term change
    delta_long = (w3 - w1)

    # Secondary signal: early change
    delta_mid = (w2 - w1)

    # Weighted combination
    trajectory_score = 0.7 * delta_long + 0.3 * delta_mid

    trajectory_score = (trajectory_score - trajectory_score.mean()) / trajectory_score.std()

    trajectory_score = trajectory_score.clip(-3, 3)

    trajectory_score = trajectory_score.replace([np.inf, -np.inf], np.nan)
    trajectory_score = trajectory_score.fillna(trajectory_score.median())

    return trajectory_score, model

from __future__ import annotations

from dataclasses import dataclass

import numpy as np
import pandas as pd

@dataclass
class TrajectoryResult:
    horizon_years: int
    mean_risk: float
    std_risk: float


def estimate_wave_drift(base: pd.DataFrame, follow: pd.DataFrame) -> pd.Series:
    base_num = base.apply(pd.to_numeric, errors="coerce")
    follow_num = follow.apply(pd.to_numeric, errors="coerce")

    common = [c for c in base_num.columns if c in follow_num.columns]
    if not common:
        return pd.Series(dtype="float64")

    drift = (follow_num[common].mean(numeric_only=True) - base_num[common].mean(numeric_only=True))
    return drift.fillna(0.0)


def simulate_forward_risk(
    base_features: pd.DataFrame,
    model,
    frailty: pd.Series,
    age: pd.Series,
    horizons=(1, 3, 5),
    n_simulations: int = 500,
    drift_scale: float = 1.0,
) -> pd.DataFrame:
    """
    Very first-pass pseudo-longitudinal simulator:
    - draws noisy perturbations around current features
    - rescored by the trained model
    - then fused with frailty and age
    """
    rng = np.random.default_rng(42)
    base = base_features.copy()

    rows = []
    feature_cols = list(base.columns)

    base_proba = model.predict_proba(base)[:, 1] if hasattr(model, "predict_proba") else model.predict(base)

    for h in horizons:
        sims = []
        for _ in range(n_simulations):
            perturbed = base.copy()

            for c in feature_cols:
                s = pd.to_numeric(perturbed[c], errors="coerce")
                noise = rng.normal(0, 0.05 * max(float(s.std(skipna=True) or 1.0), 1.0), size=len(s))
                perturbed[c] = (s + drift_scale * h * noise).astype(float)

            p = model.predict_proba(perturbed)[:, 1] if hasattr(model, "predict_proba") else model.predict(perturbed)
            fused = fuse_risk(frailty=frailty, decline_prob=pd.Series(p, index=base.index), age=age)
            sims.append(fused.values)

        sims_arr = np.vstack(sims)
        rows.append(
            {
                "horizon_years": h,
                "mean_risk": float(np.nanmean(sims_arr)),
                "std_risk": float(np.nanstd(sims_arr)),
            }
        )

    return pd.DataFrame(rows)

from __future__ import annotations

def estimate_time_to_independence_loss(
    base_features: pd.DataFrame,
    decline_model,
    frailty: pd.Series,
    age: pd.Series,
    threshold: float = 0.75,
    horizons=(1, 2, 3, 5, 7, 10),
) -> int | str:
    """
    Estimate the time to loss of independence based on pseudo-longitudinal forward simulations.
    Returns the earliest simulated horizon year where mean risk exceeds the threshold,
    or "Low long-term risk" if the threshold is never met.
    """
    sim_df = simulate_forward_risk(
        base_features=base_features,
        model=decline_model,
        frailty=frailty,
        age=age,
        horizons=horizons,
        n_simulations=500,
    )

    for _, row in sim_df.iterrows():
        if row["mean_risk"] >= threshold:
            return int(row["horizon_years"])

    return "Low long-term risk"

from __future__ import annotations

import pandas as pd
import numpy as np


def compute_decline_label(X_w1: pd.DataFrame, X_w2: pd.DataFrame) -> pd.Series:
    """
    Compute decline using domain-aligned aggregation.

    Works for HAALSI where W1 and W2 variable names differ.
    """

    # -----------------------
    # Extract domains
    # -----------------------
    def group_by_domain(df):
        domain_map = {}

        for col in df.columns:
            if "__" not in col:
                continue

            domain = col.split("__")[0]

            if domain not in domain_map:
                domain_map[domain] = []

            domain_map[domain].append(col)

        return domain_map

    w1_domains = group_by_domain(X_w1)
    w2_domains = group_by_domain(X_w2)

    common_domains = set(w1_domains.keys()).intersection(w2_domains.keys())

    if not common_domains:
        raise ValueError("No overlapping domains between W1 and W2.")

    w1_scores = []
    w2_scores = []

    # -----------------------
    # Compute per-domain scores
    # -----------------------
    for d in common_domains:

        w1_cols = w1_domains[d]
        w2_cols = w2_domains[d]

        if not w1_cols or not w2_cols:
            continue

        w1_block = X_w1[w1_cols].apply(pd.to_numeric, errors="coerce")
        w2_block = X_w2[w2_cols].apply(pd.to_numeric, errors="coerce")

        # Standardize within domain
        w1_std = (w1_block - w1_block.mean()) / w1_block.std(ddof=0)
        w2_std = (w2_block - w2_block.mean()) / w2_block.std(ddof=0)

        w1_scores.append(w1_std.mean(axis=1))
        w2_scores.append(w2_std.mean(axis=1))

    if not w1_scores:
        raise ValueError("No usable domains for decline computation.")

    # -----------------------
    # Aggregate across domains
    # -----------------------
    score_w1 = pd.concat(w1_scores, axis=1).mean(axis=1)
    score_w2 = pd.concat(w2_scores, axis=1).mean(axis=1)

    # -----------------------
    # Decline
    # -----------------------
    delta = score_w2 - score_w1

    y = (delta > 0).astype(int)

    return y

import pandas as pd
import numpy as np


def compute_domain_score(df, columns):
    """
    Aggregate a domain (e.g., mobility, cognition)
    """
    subset = df[columns].apply(pd.to_numeric, errors="coerce")
    return subset.mean(axis=1)


def compute_trajectory_score(w1_score, w2_score, w3_score):
    """
    Compute change pattern
    """
    delta1 = w2_score - w1_score
    delta2 = w3_score - w2_score

    return delta1, delta2


def assign_trajectory_class(delta1, delta2):
    """
    Rule-based trajectory classification
    """

    traj = []

    for d1, d2 in zip(delta1, delta2):
        if pd.isna(d1) or pd.isna(d2):
            traj.append("Unknown")

        elif d1 <= 0 and d2 <= 0:
            traj.append("Stable")

        elif d1 > 0 and d2 > 0:
            traj.append("Rapid Decline")

        elif d1 > 0 and d2 <= 0:
            traj.append("Recovery")

        elif d1 <= 0 and d2 > 0:
            traj.append("Late Decline")

        else:
            traj.append("Mixed")

    return pd.Series(traj)

import pandas as pd
import numpy as np


def zscore(series):
    mean = series.mean()
    std = series.std()
    if std == 0 or pd.isna(std):
        return pd.Series(0, index=series.index)
    return (series - mean) / std


def compute_domain_score(df, columns):
    """
    Standardized domain score (robust)
    """

    subset = df[columns].apply(pd.to_numeric, errors="coerce")

    # Normalize each variable
    subset = subset.apply(zscore)

    # Average
    score = subset.mean(axis=1)

    return score
